function HeroTab({ active, label, onClick }) {
  return (
    <button onClick={onClick} className={`px-8 py-3 text-sm font-bold tracking-widest border-t-2 transition-all ${active ? 'bg-[#14110e] border-[#d4af37] text-[#d4af37]' : 'bg-[#1a1612] border-transparent text-[#8b7355] hover:text-[#d1c2b1]'}`}>
      {label}
    </button>
  );
}

function SectionTitle({ title }) {
  return <h3 className="text-xs font-bold text-[#d4af37] border-b border-[#3e352a] pb-1 mb-4 uppercase tracking-wider">{title}</h3>;
}

function StatRow({ label, value, color }) {
  return (
    <div className="flex justify-between items-center text-xs">
      <span className="opacity-70">{label}</span>
      <span className="font-bold" style={{ color: color || '#d1c2b1' }}>{value}</span>
    </div>
  );
}

function StatTiny({ label, value, wide }) {
  return (
    <div className={`flex justify-between items-center text-[11px] py-1 border-b border-[#3e352a]/30 ${wide ? 'min-w-[120px]' : ''}`}>
      <span className="opacity-60">{label}:</span>
      <span className="font-bold text-[#d4af37] ml-2">{value}</span>
    </div>
  );
}

function GearSlot({ label, icon, className }) {
  return (
    <div className={`flex flex-col items-center gap-1 ${className}`}>
      <div className="w-16 h-16 border border-[#d4af37]/50 bg-[#1c1814] flex items-center justify-center text-[#d4af37]/70 hover:border-[#d4af37] hover:text-[#d4af37] cursor-pointer transition-all shadow-[inset_0_0_15px_rgba(0,0,0,0.5)] active:scale-95 group">
        {React.cloneElement(icon, { className: "group-hover:drop-shadow-[0_0_5px_#d4af37]" })}
      </div>
      <span className="text-[9px] font-bold text-[#8b7355] tracking-widest uppercase mt-1">{label}</span>
    </div>
  );
}
